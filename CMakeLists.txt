cmake_minimum_required(VERSION 3.20)
project(forge_c_open_source VERSION 0.1.0 LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(FORGE_COVERAGE "Instrument LLVM coverage on the native demo" OFF)
option(FORGE_MCDC "Instrument LLVM MC/DC for native demo" OFF)

if(FORGE_MCDC AND NOT FORGE_COVERAGE)
    message(FATAL_ERROR "FORGE_MCDC requires FORGE_COVERAGE")
endif()
if(FORGE_COVERAGE)
    if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang")
        message(FATAL_ERROR "LLVM coverage demo requires Clang compiler")
    endif()
    add_compile_options(-fprofile-instr-generate -fcoverage-mapping)
    add_link_options(-fprofile-instr-generate -fcoverage-mapping)
    if(FORGE_MCDC)
        add_compile_options(-fcoverage-mcdc)
        add_link_options(-fcoverage-mcdc)
    endif()
endif()

add_library(forge_rcc_demo STATIC drivers/demo_rcc/src/forge_rcc_demo.c)
target_include_directories(forge_rcc_demo PUBLIC drivers/demo_rcc/include)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(forge_rcc_demo PRIVATE -Wall -Wextra -Werror -Wconversion -Wsign-conversion -Wshadow -Wstrict-prototypes -fstack-usage)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(test_forge_rcc_demo tests/c/test_forge_rcc_demo.c)
    target_link_libraries(test_forge_rcc_demo PRIVATE forge_rcc_demo)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(test_forge_rcc_demo PRIVATE -Wall -Wextra -Werror -Wconversion -Wsign-conversion -Wshadow -Wstrict-prototypes -UNDEBUG)
    endif()
    add_test(NAME forge_rcc_demo_native COMMAND test_forge_rcc_demo)
endif()
