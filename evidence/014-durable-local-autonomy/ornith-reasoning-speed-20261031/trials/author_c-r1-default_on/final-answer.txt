#include <stdint.h>
#include <stddef.h>

uint32_t replace_field(uint32_t original, unsigned lsb, unsigned width, uint32_t replacement)
{
    if (width < 1u || width > 32u) return original;
    if (lsb > 31u)                 return original;
    if (lsb + width > 32u)         return original;

    uint32_t field_mask;
    uint32_t low;
    if (width == 32u) {
        field_mask = 0xFFFFFFFFu;
        low        = 0xFFFFFFFFu;
    } else {
        uint32_t bits = (1u << width) - 1u;
        field_mask    = bits << lsb;
        low           = replacement & bits;
    }

    return (original & ~field_mask) | (low << lsb);
}

int poll_match(uint32_t (*read_word)(void *), void *ctx, uint32_t mask,
               uint32_t expected, unsigned max_reads)
{
    if (read_word == NULL) return -1;

    for (unsigned i = 0u; i < max_reads; i++) {
        uint32_t word = read_word(ctx);
        if ((word & mask) == (expected & mask)) {
            return 1;
        }
    }

    return 0;
}
